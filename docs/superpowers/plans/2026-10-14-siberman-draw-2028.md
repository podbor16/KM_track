```sql
-- Жеребьёвка регистраций Siberman (с 2028). Все таблицы — по году жеребьёвки,
-- жеребьёвка 2029 не затрёт 2028. Спецификация:
-- docs/superpowers/specs/2026-10-14-siberman-draw-2028-design.md
CREATE TABLE IF NOT EXISTS draw_applicants (
  id                 INT AUTO_INCREMENT PRIMARY KEY,
  draw_year          SMALLINT NOT NULL,
  format             ENUM('individual','relay') NOT NULL,
  surname            VARCHAR(100) NOT NULL,
  name               VARCHAR(100) NOT NULL DEFAULT '',
  gender             ENUM('M','F') DEFAULT NULL,
  city               VARCHAR(100) NOT NULL DEFAULT '',
  pool               ENUM('wildcard','relay','repeat','novice') NOT NULL,
  is_wildcard        BOOLEAN NOT NULL DEFAULT FALSE,
  chances            SMALLINT NOT NULL,
  starts_individual  SMALLINT NOT NULL DEFAULT 0,
  starts_relay       SMALLINT NOT NULL DEFAULT 0,
  draw_accumulated   SMALLINT NOT NULL DEFAULT 0,
  is_local           SMALLINT NOT NULL DEFAULT 0,
  duathlon_222       SMALLINT NOT NULL DEFAULT 0,
  race_24h           SMALLINT NOT NULL DEFAULT 0,
  tattoo             SMALLINT NOT NULL DEFAULT 0,
  relay_swim         VARCHAR(200) NOT NULL DEFAULT '',
  relay_bike         VARCHAR(200) NOT NULL DEFAULT '',
  relay_run          VARCHAR(200) NOT NULL DEFAULT '',
  wildcard_reason    TEXT,
  results_text       TEXT,
  payment_status     ENUM('none','pending','paid','unpaid') NOT NULL DEFAULT 'none',
  payment_comment    VARCHAR(500) NOT NULL DEFAULT '',
  payment_updated_at DATETIME DEFAULT NULL,
  UNIQUE KEY uq_draw_person (draw_year, format, surname, name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- version увеличивается каждой записью — по нему публичный кэш понимает, что
-- снимок устарел. Строка года блокируется (FOR UPDATE) на время каждой записи.
CREATE TABLE IF NOT EXISTS draw_config (
  draw_year          SMALLINT PRIMARY KEY,
  capacity_men       SMALLINT DEFAULT NULL,
  capacity_women     SMALLINT DEFAULT NULL,
  capacity_relay     SMALLINT DEFAULT NULL,
  quota_repeat_men   SMALLINT DEFAULT NULL,
  quota_repeat_women SMALLINT DEFAULT NULL,
  quota_novice_men   SMALLINT DEFAULT NULL,
  quota_novice_women SMALLINT DEFAULT NULL,
  current_round      TINYINT NOT NULL DEFAULT 0,
  auto_invite_until  DATE DEFAULT NULL,
  imported_at        DATETIME DEFAULT NULL,
  version            INT NOT NULL DEFAULT 0
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Журнал вытягиваний. Порядок в зачёте — position_key (у новой записи = её id,
-- у исправления — position_key заменённой). Отмена/исправление не удаляют строку.
CREATE TABLE IF NOT EXISTS draw_picks (
  id            INT AUTO_INCREMENT PRIMARY KEY,
  draw_year     SMALLINT NOT NULL,
  applicant_id  INT NOT NULL,
  round         TINYINT NOT NULL,
  category      ENUM('men','women','relay') NOT NULL,
  position_key  INT NOT NULL DEFAULT 0,
  created_at    DATETIME NOT NULL,
  cancelled_at  DATETIME DEFAULT NULL,
  replaced_by   INT DEFAULT NULL,
  KEY ix_draw_picks_year (draw_year),
  FOREIGN KEY (applicant_id) REFERENCES draw_applicants(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS draw_audit (
  id               INT AUTO_INCREMENT PRIMARY KEY,
  draw_year        SMALLINT NOT NULL,
  applicant_id     INT NOT NULL,
  action           VARCHAR(30) NOT NULL,
  old_value        VARCHAR(200) NOT NULL DEFAULT '',
  new_value        VARCHAR(200) NOT NULL DEFAULT '',
  ref_applicant_id INT DEFAULT NULL,
  admin_user       VARCHAR(100) NOT NULL DEFAULT '',
  created_at       DATETIME NOT NULL,
  KEY ix_draw_audit_year (draw_year)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
```

```python
"""
Таблица организаторов жеребьёвки (лист «Спец вставка») → заявки draw_applicants.

Колонки ищутся по заголовку. Пул (цвет шарика): wild card (красные) → 'wildcard',
эстафета (синие) → 'relay', личник со стартами в личном зачёте → 'repeat',
иначе → 'novice' (чёрные, две ёмкости). Отметки «!» и «Участвует 2027» не
используются: решение организаторов — обычные участники (2026-10-14).
Эстафета — по капитану, названия команды в таблице нет.
"""
import io

import openpyxl

SHEET = "Спец вставка"
COL_FORMAT, COL_NAME, COL_SEX, COL_CITY = "format", "surname Name", "sex", "city (исправлено)"
COL_WILDCARD, COL_TOTAL, COL_EXTRA = "Wildcard", "Общая сумма шансов", "Сумма дополнительных шансов"
COL_WILDCARD_REASON, COL_RESULTS = "wildcard_reason", "Расскажите_о_ваших_результатах"
EXTRA_COLS = {
    "starts_individual": "Старты на гонке (индивидуально)",
    "starts_relay": "Старты на гонке (эстафета)",
    "draw_accumulated": "Накопленные шансы за участия в жеребьёвке",
    "is_local": "Карточка для местных атлетов",
    "duathlon_222": "Участие в 222 Дуатлоне",
    "race_24h": "Участие в суточной велогонке 24часа",
    "tattoo": "Тату Siberman",
}
REQUIRED = (COL_FORMAT, COL_NAME, COL_SEX, COL_CITY, COL_WILDCARD, COL_TOTAL, COL_EXTRA, *EXTRA_COLS.values())
RELAY_ROLES = ("swim", "bike", "run")


def _norm(v) -> str:
    return " ".join(str(v if v is not None else "").split())


def _int(v) -> int:
    if v in (None, ""):
        return 0
    if isinstance(v, (int, float)) and float(v).is_integer():
        return int(v)
    if isinstance(v, str) and v.strip().isdigit():
        return int(v.strip())
    raise ValueError(f"не число: «{v}»")


def parse_applicants(file_bytes: bytes) -> tuple[list[dict], list[str]]:
    """(заявки, ошибки). Строка с ошибкой в заявки не попадает."""
    wb = openpyxl.load_workbook(io.BytesIO(file_bytes), data_only=True)
    ws = wb[SHEET] if SHEET in wb.sheetnames else wb.worksheets[0]
    rows = list(ws.iter_rows(values_only=True))
    if not rows:
        return [], ["пустой лист"]
    idx: dict[str, int] = {}
    for i, h in enumerate(rows[0]):
        if _norm(h) and _norm(h) not in idx:
            idx[_norm(h)] = i
    missing = [c for c in REQUIRED if c not in idx]
    if missing:
        return [], ["нет колонок: " + ", ".join(missing)]

    apps: list[dict] = []
    errors: list[str] = []
    seen: set[tuple] = set()
    for n, row in enumerate(rows[1:], start=2):
        def get(col, _row=row):
            i = idx.get(col)
            return _row[i] if i is not None and i < len(_row) else None

        full = _norm(get(COL_NAME))
        if not full:
            continue
        where = f"строка {n} «{full}»"
        fmt_raw = _norm(get(COL_FORMAT))
        if fmt_raw.startswith("Эстафет"):
            fmt = "relay"
        elif fmt_raw.startswith("Индивид"):
            fmt = "individual"
        else:
            errors.append(f"{where}: формат «{fmt_raw}»")
            continue
        gender = {"Мужской": "M", "Женский": "F"}.get(_norm(get(COL_SEX)))
        if fmt == "individual" and gender is None:
            errors.append(f"{where}: пол «{_norm(get(COL_SEX))}»")
            continue
        try:
            extra = {k: _int(get(col)) for k, col in EXTRA_COLS.items()}
            total, extra_sum = _int(get(COL_TOTAL)), _int(get(COL_EXTRA))
        except ValueError as e:
            errors.append(f"{where}: {e}")
            continue
        if sum(extra.values()) != extra_sum or total != extra_sum + 1:
            errors.append(f"{where}: шансы не сходятся (доп. {sum(extra.values())}, "
                          f"«сумма доп.» {extra_sum}, итого {total})")
            continue
        surname, _, name = full.partition(" ")
        key = (fmt, surname.lower(), name.lower())
        if key in seen:
            errors.append(f"{where}: повтор заявки")
            continue
        seen.add(key)
        is_wildcard = _norm(get(COL_WILDCARD)) == "Wildcard"
        if is_wildcard:
            pool = "wildcard"
        elif fmt == "relay":
            pool = "relay"
        else:
            pool = "repeat" if extra["starts_individual"] > 0 else "novice"
        relay = {
            f"relay_{r}": _norm(f"{_norm(get('surname_' + r))} {_norm(get('name_' + r))}") if fmt == "relay" else ""
            for r in RELAY_ROLES
        }
        apps.append({
            "format": fmt, "surname": surname, "name": name,
            "gender": gender if fmt == "individual" else None,
            "city": _norm(get(COL_CITY)), "pool": pool, "is_wildcard": is_wildcard,
            "chances": total, **extra, **relay,
            "wildcard_reason": _norm(get(COL_WILDCARD_REASON)), "results_text": _norm(get(COL_RESULTS)),
        })
    return apps, errors


def import_report(apps: list[dict]) -> dict:
    """Люди и шарики по группам: repeat_M, novice_F, relay, wildcard_M, wildcard (эстафеты)…"""
    groups: dict[str, dict] = {}
    for a in apps:
        key = a["pool"] if a["format"] == "relay" else f"{a['pool']}_{a['gender']}"
        g = groups.setdefault(key, {"people": 0, "balls": 0})
        g["people"] += 1
        g["balls"] += a["chances"]
    return {"total": len(apps), "groups": groups}
```

```python
"""
Чистая логика жеребьёвки: пулы раундов, списки зачётов, проверка вытягивания, счётчики.

Раунды (ответы пользователя 2026-10-13): 0 — wild card (красные шарики), 1 — эстафеты
(синие), 2 — повторные личники (чёрные, ёмкость 1), 3 — новички (чёрные, ёмкость 2),
4 — все оставшиеся личники (обе ёмкости вместе). Мужчины и женщины одного цвета —
зачёт определяется по полу. Порядок в зачёте — порядок вытягивания (position_key);
№1…мест — «победитель», дальше — «лист ожидания».
"""
CATEGORIES = ("men", "women", "relay")
CATEGORY_LABELS = {"men": "Мужчины", "women": "Женщины", "relay": "Эстафеты"}
ROUND_LABELS = {0: "Wild card", 1: "Эстафеты", 2: "Повторные", 3: "Новички", 4: "Все личники"}
POOL_LABELS = {"wildcard": "wild card", "relay": "эстафета", "repeat": "повторный", "novice": "новичок"}


def category_of(app: dict) -> str:
    if app["format"] == "relay":
        return "relay"
    return "men" if app["gender"] == "M" else "women"


def kind_of(app: dict) -> str:
    """Повторный/новичок — по стартам в личном зачёте; wild card тоже засчитывается в квоту."""
    return "repeat" if app["starts_individual"] > 0 else "novice"


def in_round_pool(app: dict, rnd: int) -> bool:
    if rnd == 0:
        return app["is_wildcard"]
    if app["is_wildcard"]:
        return False
    if rnd == 1:
        return app["pool"] == "relay"
    if rnd == 2:
        return app["pool"] == "repeat"
    if rnd == 3:
        return app["pool"] == "novice"
    if rnd == 4:
        return app["pool"] in ("repeat", "novice")
    return False


def build_lists(apps_by_id: dict, picks: list[dict], config: dict) -> dict[str, list[dict]]:
    lists: dict[str, list[dict]] = {c: [] for c in CATEGORIES}
    active = sorted((p for p in picks if p["cancelled_at"] is None),
                    key=lambda p: (p["position_key"], p["id"]))
    for p in active:
        a = apps_by_id[p["applicant_id"]]
        lists[p["category"]].append({
            "pick_id": p["id"], "position_key": p["position_key"], "applicant_id": a["id"],
            "surname": a["surname"], "name": a["name"], "city": a["city"],
            "is_wildcard": a["is_wildcard"], "round": p["round"], "created_at": p["created_at"],
            "payment_status": a["payment_status"],
        })
    for c, rows in lists.items():
        cap = config.get(f"capacity_{c}")
        for i, row in enumerate(rows, start=1):
            row["position"] = i
            row["section"] = None if cap is None else ("winner" if i <= cap else "waitlist")
    return lists


def placement(lists: dict, applicant_id: int) -> tuple[str, int] | None:
    for c, rows in lists.items():
        for r in rows:
            if r["applicant_id"] == applicant_id:
                return c, r["position"]
    return None


def check_pick(app: dict, rnd: int, lists: dict) -> str | None:
    """Причина отказа или None, если участника можно вытянуть в раунде rnd."""
    placed = placement(lists, app["id"])
    if placed:
        return f"уже вытянут — {CATEGORY_LABELS[placed[0]]} №{placed[1]}"
    if not in_round_pool(app, rnd):
        return f"не в этом раунде: {POOL_LABELS[app['pool']]}"
    return None


def round_counters(apps: list[dict], lists: dict, rnd: int) -> dict:
    by_id = {a["id"]: a for a in apps}
    kinds = {f"{k}_{g}": 0 for k in ("repeat", "novice") for g in ("M", "F")}
    for c in ("men", "women"):
        for r in lists[c]:
            a = by_id[r["applicant_id"]]
            kinds[f"{kind_of(a)}_{a['gender']}"] += 1
    placed = {r["applicant_id"] for rows in lists.values() for r in rows}
    pool = [a for a in apps if a["id"] not in placed and in_round_pool(a, rnd)]
    return {
        "categories": {c: len(rows) for c, rows in lists.items()},
        "kinds": kinds,
        "pool_people": len(pool),
        "pool_balls": sum(a["chances"] for a in pool),
    }
```

```python
"""
Вероятность попасть в победители — Monte Carlo от текущего состояния жеребьёвки.

Шарик с весом = шансы, после выигрыша остальные шарики человека не участвуют =
взвешенная выборка людей без возвращения (Efraimidis–Spirakis: ключ random()**(1/вес),
по убыванию). Ещё не объявленные wild card — гарантированные места (сами в результат
не попадают: публично их нет до объявления).

Без numpy (решение 2026-10-14): ~0,5–0,8 с на 240 участников × 5000 прогонов, считается
в фоне раз на изменение. Выборка — только в _weighted_order (заменяема на numpy).
"""
import random

from src.siberman.draw.logic import category_of, kind_of

REQUIRED_CONFIG = ("capacity_men", "capacity_women", "capacity_relay", "quota_repeat_men",
                   "quota_repeat_women", "quota_novice_men", "quota_novice_women")
GENDER_CATEGORY = {"M": "men", "F": "women"}


def _weighted_order(people: list[tuple[int, int]], rnd) -> list[int]:
    return [pid for _, pid in sorted(((rnd() ** (1.0 / w), pid) for pid, w in people), reverse=True)]


def win_probabilities(apps: list[dict], lists: dict, config: dict, current_round: int,
                      sims: int = 5000, seed=None) -> dict[int, float] | None:
    """{applicant_id: P(победитель)} для ещё не вытянутых не-wild card; None — места/квоты не заданы."""
    if any(config.get(k) is None for k in REQUIRED_CONFIG):
        return None
    rnd = random.Random(seed).random
    by_id = {a["id"]: a for a in apps}
    cap = {c: config[f"capacity_{c}"] for c in ("men", "women", "relay")}
    quota = {f"{k}_{g}": config[f"quota_{k}_{GENDER_CATEGORY[g]}"] for k in ("repeat", "novice") for g in ("M", "F")}
    placed = {r["applicant_id"] for rows in lists.values() for r in rows}

    count0 = {c: len(lists[c]) for c in cap}
    kind0 = dict.fromkeys(quota, 0)
    for c in ("men", "women"):
        for r in lists[c]:
            a = by_id[r["applicant_id"]]
            kind0[f"{kind_of(a)}_{a['gender']}"] += 1
    for a in apps:
        if a["is_wildcard"] and a["id"] not in placed:
            c = category_of(a)
            count0[c] += 1
            if c != "relay":
                kind0[f"{kind_of(a)}_{a['gender']}"] += 1

    undrawn = [a for a in apps if a["id"] not in placed and not a["is_wildcard"]]
    relay = [(a["id"], a["chances"]) for a in undrawn if a["pool"] == "relay"]
    by_kind = {k: [(a["id"], a["chances"]) for a in undrawn if a["pool"] == k] for k in ("repeat", "novice")}
    wins = dict.fromkeys((a["id"] for a in undrawn), 0)

    for _ in range(sims):
        count, kinds = dict(count0), dict(kind0)
        if current_round <= 1:
            free = max(0, cap["relay"] - count["relay"])
            for pid in _weighted_order(relay, rnd)[:free]:
                wins[pid] += 1
        leftovers: list[int] = []
        for round_no, kind in ((2, "repeat"), (3, "novice")):
            people = by_kind[kind]
            if current_round > round_no:
                leftovers.extend(pid for pid, _ in people)
                continue
            for pid in _weighted_order(people, rnd):
                g = by_id[pid]["gender"]
                c = GENDER_CATEGORY[g]
                if kinds[f"{kind}_{g}"] < quota[f"{kind}_{g}"] and count[c] < cap[c]:
                    wins[pid] += 1
                    kinds[f"{kind}_{g}"] += 1
                    count[c] += 1
                else:
                    leftovers.append(pid)
        if count["men"] >= cap["men"] and count["women"] >= cap["women"]:
            continue
        for pid in _weighted_order([(pid, by_id[pid]["chances"]) for pid in leftovers], rnd):
            c = GENDER_CATEGORY[by_id[pid]["gender"]]
            if count[c] < cap[c]:
                wins[pid] += 1
                count[c] += 1
    return {pid: w / sims for pid, w in wins.items()}
```

```python
def _snapshot(conn, year: int) -> dict:

def _applicant(snap: dict, applicant_id: int) -> dict:
    a = next((x for x in snap["applicants"] if x["id"] == applicant_id), None)

def _write(year: int, op):

    conn = _connect()
    try:
        ddb.ensure_config(conn, year)
        conn.start_transaction()
        ddb.lock_config(conn, year)
        result = op(conn, _snapshot(conn, year))
        ddb.bump_version(conn, year)
        conn.commit()
        return result
    except Exception:
        conn.rollback()
        raise
    finally:
        conn.close()

def pick(year: int, applicant_id: int, user: str = "") -> dict:

            raise DrawError(f"{_fio(a)}: другой зачёт — используйте «Отменить»")

            raise DrawError(f"{_fio(a)}: {reason}")

        ddb.audit(conn, year, a["id"], "replace_pick", str(p["applicant_id"]), str(a["id"]), p["applicant_id"], user)

        if old == "unpaid" and status != "unpaid":
            invited_id = ddb.last_auto_invite(conn, year, a["id"])
            invited = next((x for x in snap["applicants"] if x["id"] == invited_id), None)
            if invited and invited["payment_status"] == "pending":
                result["suggest_revert"] = {"applicant_id": invited["id"], "surname": invited["surname"],
                                            "name": invited["name"]}
        return result
    return _write(year, op)

def admin_state(year: int) -> dict:
    conn = _connect()
    try:
        snap = _snapshot(conn, year)
    finally:
        conn.close()
    cfg, rnd = snap["config"], snap["config"]["current_round"]
    by_id = {a["id"]: a for a in snap["applicants"]}
    positions = {r["pick_id"]: r["position"] for rows in snap["lists"].values() for r in rows}
    for a in snap["applicants"]:
        placed = placement(snap["lists"], a["id"])
        a["placed"] = {"category": placed[0], "position": placed[1]} if placed else None
        a["in_round"] = in_round_pool(a, rnd)
        a["category"] = category_of(a)
    journal = [{
        "pick_id": p["id"], "applicant_id": p["applicant_id"],
        "surname": by_id[p["applicant_id"]]["surname"], "name": by_id[p["applicant_id"]]["name"],
        "round": p["round"], "category": p["category"], "created_at": p["created_at"],
        "cancelled_at": p["cancelled_at"], "replaced_by": p["replaced_by"], "position": positions.get(p["id"]),
    } for p in sorted(snap["picks"], key=lambda p: p["id"], reverse=True)]
    return {"year": year, "config": cfg, "round_labels": ROUND_LABELS, "category_labels": CATEGORY_LABELS,
            "applicants": snap["applicants"], "lists": snap["lists"], "journal": journal,
            "counters": round_counters(snap["applicants"], snap["lists"], rnd)}


def applicant_summary(year: int, applicant_id: int) -> dict:
    conn = _connect()
    try:
        snap = _snapshot(conn, year)
        a = _applicant(snap, applicant_id)
        history = ddb.siberman_history(conn, a["surname"], a["name"])
        audit = ddb.load_audit(conn, year, applicant_id)
    finally:
        conn.close()
    placed = placement(snap["lists"], applicant_id)
    where = None
    if placed:
        row = next(r for r in snap["lists"][placed[0]] if r["applicant_id"] == applicant_id)
        where = {"category": placed[0], "position": placed[1], "round": row["round"], "created_at": row["created_at"]}
    return {"applicant": a, "history": history, "placement": where, "audit": audit}


# ── Чтение: публика (кэш) ───────────────────────────────────────────────────

def _cached_snapshot(year: int) -> dict:
    now = time.monotonic()
    hit = _cache.get(year)
    if hit and now - hit[1] < CACHE_TTL_S:
        return hit[2]
    conn = _connect()
    try:
        version = ddb.get_version(conn, year)
        if hit and hit[0] == version:
            snap = hit[2]
        else:
            snap = _snapshot(conn, year)
            snap["version"] = version
    finally:
        conn.close()
    _cache[year] = (version, now, snap)
    _ensure_probabilities(snap)
    return snap


def _ensure_probabilities(snap: dict) -> None:
    """Пересчёт вероятностей в фоне при смене версии; публика видит последний готовый."""
    year, version = snap["year"], snap["version"]
    with _lock:
        done = _probs.get(year)
        if (done and done[0] == version) or year in _probs_running:
            return
        _probs_running.add(year)

    def run():
        try:
            probs = win_probabilities(snap["applicants"], snap["lists"], snap["config"],
                                      snap["config"]["current_round"], sims=SIMS)
            with _lock:
                _probs[year] = (version, probs)
        finally:
            with _lock:
                _probs_running.discard(year)

    threading.Thread(target=run, daemon=True).start()


def _last_row(snap: dict) -> dict | None:
    active = [p for p in snap["picks"] if p["cancelled_at"] is None]
    if not active:
        return None
    last_id = max(p["id"] for p in active)
    for c, rows in snap["lists"].items():
        for r in rows:
            if r["pick_id"] == last_id:
                return {**r, "category": c}
    return None


def _aquarium_group(a: dict, rnd: int) -> str:
    if a["format"] == "relay":
        return "relay"
    return "individual" if rnd >= 4 else a["pool"]


def public_state(year: int) -> dict:
    """Публично: без признака wild card и без необъявленных wild card (их нет нигде)."""
    snap = _cached_snapshot(year)
    cfg, rnd = snap["config"], snap["config"]["current_round"]
    done = _probs.get(year)
    probs = done[1] if done else None
    placed = {r["applicant_id"] for rows in snap["lists"].values() for r in rows}
    people = [a for a in snap["applicants"] if not a["is_wildcard"] and a["id"] not in placed]
    group_keys = ["relay"] + (["individual"] if rnd >= 4 else ["repeat", "novice"])
    groups = []
    for key in group_keys:
        members = [a for a in people if _aquarium_group(a, rnd) == key]
        label, color = AQUARIUM_GROUPS[key]
        groups.append({"key": key, "label": label, "color": color, "people": len(members),
                       "balls": sum(a["chances"] for a in members)})
    order = {k: i for i, k in enumerate(group_keys)}
    people.sort(key=lambda a: (order[_aquarium_group(a, rnd)], -a["chances"], a["surname"], a["name"]))
    last = _last_row(snap)
    return {
        "year": year, "round": rnd, "round_label": ROUND_LABELS[rnd],
        "capacities": {c: cfg[f"capacity_{c}"] for c in CATEGORIES},
        "lists": {c: [{
            "position": r["position"], "surname": r["surname"], "name": r["name"], "city": r["city"],
            "section": r["section"], "status": PAYMENT_ICONS[r["payment_status"]],
            "is_last": last is not None and r["pick_id"] == last["pick_id"],
        } for r in rows] for c, rows in snap["lists"].items()},
        "aquarium": {"groups": groups, "people": [{
            "surname": a["surname"], "name": a["name"], "city": a["city"], "chances": a["chances"],
            "group": _aquarium_group(a, rnd), "probability": None if probs is None else probs.get(a["id"]),
        } for a in people]},
    }


# ── Фиды vMix ───────────────────────────────────────────────────────────────

def feed_rows(year: int, name: str) -> list[dict]:
    snap = _cached_snapshot(year)
    cfg = snap["config"]
    if name in CATEGORIES:
        cap = cfg[f"capacity_{name}"]
        rows = snap["lists"][name]
        out = [{"num": r["position"], "name": _fio(r), "city": r["city"],
                "section": SECTION_LABELS[r["section"]], "status": PAYMENT_ICONS[r["payment_status"]]}
               for r in rows]
        for i in range(len(rows) + 1, (cap or 0) + 1):
            out.append({"num": i, "name": "", "city": "", "section": SECTION_LABELS["winner"], "status": ""})
        return out
    if name == "wildcard":
        wc = sorted(({**r, "category": c} for c, rows in snap["lists"].items() for r in rows if r["is_wildcard"]),
                    key=lambda r: r["position_key"])
        return [{"num": i, "name": _fio(r), "city": r["city"], "category": CATEGORY_LABELS[r["category"]]}
                for i, r in enumerate(wc, start=1)]
    if name == "last":
        last = _last_row(snap)
        if last is None:
            return []
        return [{"name": _fio(last), "city": last["city"], "category": CATEGORY_LABELS[last["category"]],
                 "num": last["position"], "round": ROUND_LABELS[last["round"]]}]
    raise DrawError("нет такого фида")


def to_csv(rows: list[dict], columns: tuple) -> str:
    buf = io.StringIO()
    writer = csv.DictWriter(buf, fieldnames=list(columns), lineterminator="\n")
    writer.writeheader()
    writer.writerows(rows)
    return "\ufeff" + buf.getvalue()
```

```javascript
function inRound(a, rnd) { // зеркало logic.in_round_pool
  if (rnd === 0) return a.is_wildcard;
  if (a.is_wildcard) return false;
  return ({ 1: a.pool === 'relay', 2: a.pool === 'repeat', 3: a.pool === 'novice', 4: a.pool === 'repeat' || a.pool === 'novice' })[rnd] || false;
}

function reasonFor(a) {
  if (a.placed) return `уже вытянут — ${catLabel(a.placed.category)} №${a.placed.position}`;
  if (replacing) {
    if (a.category !== replacing.category) return `другой зачёт: ${catLabel(a.category)}`;
    if (!inRound(a, replacing.round)) return 'не из раунда исправляемой записи';
    return '';
  }
  if (!a.in_round) return `не в этом раунде: ${POOL[a.pool]}`;
  return '';
}

function candidates() {

  if (!q) return [];
  return S.applicants

    .sort((x, y) => (x.reason ? 1 : 0) - (y.reason ? 1 : 0) || fio(x.a).localeCompare(fio(y.a), 'ru'))
    .slice(0, 12);
}

  else if (e.key === 'Escape') { search.value = ''; cancelReplace(); }
});

load();
// Оператор видит изменения директора (оплаты) и второй вкладки; другие вкладки не перерисовываем,
```

```javascript
// Публичные блоки жеребьёвки (iframe в Тильде): опрос состояния и высота для страницы Тильды.
const DRAW_POLL_MS = 3000;

function drawEscape(s) {
  return String(s ?? '').replace(/[&<>"]/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]));
}

function drawPoll(year, render) {
  let lastBody = null;
  async function tick() {
    try {
      const r = await fetch(`/api/siberman/draw/${year}/state`, { cache: 'no-store' });
      if (r.ok) {
        const body = await r.text();
        if (body !== lastBody) { lastBody = body; render(JSON.parse(body)); }
      }
    } catch (e) { /* сеть моргнула — следующий опрос */ }
    setTimeout(tick, DRAW_POLL_MS);
  }
  tick();
}

// Высота содержимого → странице Тильды (скрипт вставки — docs/siberman/draw-tilda-snippet.md)
(function () {
  if (window.parent === window) return;
  const send = () => window.parent.postMessage({ sibermanDrawHeight: document.documentElement.scrollHeight }, '*');
  new ResizeObserver(send).observe(document.body);
  window.addEventListener('load', send);
})();
```
